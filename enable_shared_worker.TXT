def main():
    prepared = []

    # Validate all changes before writing anything.
    for path, replacements in PATCHES:
        original = path.read_text(encoding="utf-8-sig")
        updated = original

        for old, new in replacements:
            old_count = updated.count(old)
            new_count = updated.count(new)

            if new_count == 1 and old_count == 0:
                continue

            if old_count != 1 or new_count != 0:
                raise RuntimeError(
                    f"Unexpected source layout in {path.name}.\n"
                    "No files have been changed. Paste this error "
                    "rather than editing the engine manually."
                )

            updated = updated.replace(old, new, 1)

        compile(updated, str(path), "exec")

        if updated != original:
            prepared.append((path, updated))

    if not prepared:
        print("Shared-worker support is already installed.")
        return

    backup_root = (
        ROOT / "backups"
        / datetime.now().strftime("shared_worker_%Y%m%d_%H%M%S_%f")
    )

    # Back up every affected file before replacing any of them.
    for path, _ in prepared:
        backup = backup_root / path.relative_to(ROOT)
        backup.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(path, backup)

    print("Backups:", backup_root)

    for path, updated in prepared:
        temporary = path.with_name(path.name + ".v04.tmp")
        temporary.write_text(updated, encoding="utf-8")
        temporary.replace(path)
        print("Updated:", path.relative_to(ROOT))

    print("Shared-worker support installed.")


if __name__ == "__main__":
    main()
